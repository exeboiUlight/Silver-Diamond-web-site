Silver Diamond Game Engine 0.2

Обзор движка

Silver Diamond Game Engine — компактный кроссплатформенный движок на чистом C99. Один шейдер для всего 3D, zero-copy загрузка glTF, свой сканер JSON, атлас шрифтов, 2D/3D физика и spatial-аудио — без внешних зависимостей, кроме OpenGL и оконного слоя.

Ключевые принципы

ПринципКак реализовано
Единый шейдер для 3DForward PBR-lite: ambient + sun + до 8 point lights, скиннинг, нормал-маппинг, alphaMode OPAQUE/MASK/BLEND
Zero-copy glTFДанные вершин читаются прямо из бинарного чанка — GPU-загрузка без промежуточных копий
Одна арена на модельЗагрузка = O(#блоков) malloc'ов, освобождение — один вызов

ДетерминизмФизика не зависит от порядка вызовов — можно тестировать без окна
Один draw call для UIВсе прямоугольники и глифы батчатся в один VBO, SDF-шейдер

Платформы и требования

  • Windows: Win32 + WGL, MinGW или MSVC
  • Linux: X11 + GLX, GCC/Clang
  • OpenGL: 3.3+ (core profile)
  • Аудио: waveOut (Win) / ALSA (Linux) — опционально, без устройства движок работает молча
Один заголовок Подключите #include "sdge.h" — он включает весь публичный API движка.

Установка и сборка

Движок поддерживает три режима: динамическая библиотека, статическая сборка и header-only для отдельных модулей.

Режим 1: динамическая библиотека

Соберите движок как sdge.dll (или libsdge.so):

# Движок
cc -DSDGE_BUILD_DLL -c *.c
cc -shared -o sdge.dll *.o

# Приложение
cc -DSDGE_USE_DLL main.c -L. -lsdge

Режим 2: статическая сборка

Скомпилируйте все .c движка вместе с приложением — без SDGE_BUILD_DLL и SDGE_USE_DLL:

cc main.c engine/*.c -lglfw -lGL -o game

Режим 3: header-only модули

Некоторые модули живут прямо в заголовках. Активируйте их реализацию в одном translation unit:

// engine_impl.c
#define SDGE_WINDOW_IMPLEMENTATION
#define SDGE_ASSETS_IMPLEMENTATION
#define SDGE_GLTF_IMPLEMENTATION
#include "sdge.h"

Флаги компиляции

ФлагНазначение
SDGE_BUILD_DLLЭкспортировать символы (сборка DLL)
SDGE_USE_DLLИмпортировать символы (использование DLL)
SDGE_NO_GLОтключить OpenGL — для headless-тестов. sdgr_*, sdui_*, sdge_mesh_* вернут заглушки
SDGE_NO_ALSALinux без ALSA (silent device)

Быстрый старт

Полноценный пример: окно, камера, загрузка glTF-модели с анимацией, отрисовка. 60 строк — и всё работает.

#include "sdge.h"

int main(void) {
    // 1. Окно и контекст
    sdge_init();
    sdge_window *w = sdge_create_window(1280, 720, "Hello SDGE", NULL);
    if (!w) return 1;
    sdge_make_context_current(w);

    // 2. Инициализация рендерера и UI
    sdgr_init();
    sdui_init();
    sdge_audio_init();

    // 3. Камера
    sdge_camera3d cam;
    sdge_camera3d_init(&cam, 55.0f, 1280.0f/720.0f, 0.1f, 200.0f);
    sdge_camera3d_orbit(&cam, (float[3]){0,0,0}, 5.0f, 0.6f, 0.35f);

    // 4. Модель и анимация
    sdge_model *hero = sdge_model_load("assets/hero.glb");
    if (hero && sdge_model_anim_count(hero) > 0)
        sdge_model_play(hero, 0, 1);   // animation 0, loop

    // 5. Главный цикл
    double prev = sdge_get_time();
    while (!sdge_window_should_close(w)) {
        sdge_poll_events();

        double now = sdge_get_time();
        float dt = (float)(now - prev);
        prev = now;

        sdge_camera3d_update(&cam);
        sdge_model_advance(hero, dt);

        // 3D-проход
        sdgr_resize(1280, 720);
        sdgr_begin_frame(NULL);
        sdgr_set_camera(cam.view, cam.proj, cam.pos);
        sdgr_set_ambient(0.2f, 0.2f, 0.25f);
        sdgr_set_sun((float[3]){-0.5f,-1.0f,-0.3f}, (float[3]){1,1,1}, 2.0f);
        sdge_model_draw(hero, NULL);
        sdgr_end_frame();

        // UI-проход
        sdui_begin(1280, 720);
        sdui_text(20, 20, "Hello, SDGE", 20.0f, (float[4]){1,1,1,1});
        sdui_end();

        sdge_swap_buffers(w);
    }

    // 6. Очистка
    sdge_model_destroy(hero);
    sdui_shutdown();
    sdgr_shutdown();
    sdge_audio_terminate();
    sdge_destroy_window(w);
    sdge_terminate();
    return 0;
}

Окно и ввод

GLFW-подобный слой. Header-only, работает на Windows (Win32/WGL) и Linux (X11/GLX).

Жизненный цикл

bool sdge_init(void);
void sdge_terminate(void);
Инициализация и завершение — идемпотентны. sdge_terminate закрывает все окна.

Создание окна

sdge_window* sdge_create_window(int w, int h, const char *title, sdge_window *share);
void sdge_destroy_window(sdge_window *w);
Создаёт окно с OpenGL-контекстом. share — другое окно, с которым разделяются GL-ресурсы (или NULL).

Управление окном

  • sdge_show_window / sdge_hide_window
  • sdge_set_window_title
  • sdge_set_window_size / sdge_get_window_size — клиентская область
  • sdge_get_framebuffer_size — backbuffer (учитывает HiDPI)
  • sdge_set_fullscreen / sdge_is_fullscreen
  • sdge_swap_buffers — после конца кадра
  • sdge_swap_interval(1) — включить vsync

Таймеры

Секунды с первого вызова. Основано на QPC (Win) / CLOCK_MONOTONIC (Linux).

Математика

Header-only math3d.h: column-major mat4, right-handed, −Z forward, +Y up. Всё, что нужно для 3D.

Типы

ТипФормат
sdg_vec3float[3]
sdg_vec4float[4]
sdg_quatfloat[4] — (x, y, z, w), w = cos(θ/2)
sdg_mat4float[16], column-major: index = col*4 + row

vec3

sdg_vec3 a = {1, 2, 3}, b = {4, 5, 6}, r;

v3_add(r, a, b);          // r = a + b
v3_sub(r, a, b);          // r = a - b
v3_mul(r, a, 2.0f);       // r = a * 2
v3_cross(r, a, b);        // r = a × b
float d = v3_dot(a, b);
float l = v3_norm(r, a);  // r = normalize(a), возвращает длину
v3_lerp(r, a, b, 0.5f);

Frustum culling

glTF-модели

Загрузчик .glb и .gltf + .bin. Всё в одной арене, доступ к данным — zero-copy.

Загрузка

sdge_gltf_model *m = sdge_load_gltf("assets/character.glb");
if (!m) return 1;

// ... использовать ...

sdge_gltf_destroy(m);   // освобождает всю модель одним вызовом

Структура модели

typedef struct {
    sdge_gltf_node *nodes;      int node_count;
    int *roots;                 int root_count;
    sdge_gltf_mesh *meshes;     int mesh_count;
    sdge_gltf_skin *skins;      int skin_count;
    sdge_gltf_animation *anims; int anim_count;
    sdge_gltf_material *materials; int material_count;
    // ...текстуры, сэмплеры, bufferViews
    sdga_arena arena;           // владеет всей памятью
} sdge_gltf_model;

Узлы (nodes)

for (int i = 0; i < m->node_count; ++i) {
    const sdge_gltf_node *n = &m->nodes[i];
    if (n->mesh >= 0) {
        // у ноды есть меш
        printf("node %s → mesh %d\n", n->name, n->mesh);
    }
}

Изображения

Загрузка

if (sdga_image_load_rgba("assets/wood.png", &img) == 0) {
    // img.pixels — RGBA8, img.w * img.h * 4 байт
    // top row first
    sdga_image_free(&img);
}

Загрузка на GPU

Загружает текстуру в OpenGL и возвращает GL name. Вызывающий владеет результатом:

unsigned tex = sdgr_upload_texture(img.w, img.h, img.pixels);

// Полный контроль:
sdgr_texture_desc desc = {0};
desc.srgb       = 1;                          // sRGB для color maps
desc.wrap_s     = SDGR_REPEAT;
desc.wrap_t     = SDGR_REPEAT;
desc.min_filter = SDGR_LINEAR_MIPMAP_LINEAR;
desc.mag_filter = SDGR_LINEAR;
desc.mipmaps    = 1;

unsigned tex = sdgr_upload_texture_ex(img.w, img.h, img.pixels, &desc);

sRGB vs linear

OcclusionLinear

Шрифты

Загрузчик TTF и OTF (CFF) с растеризатором в атлас. Плюс встроенный 5×7 без файла.

Загрузка TTF/OTF

sdga_font *f = sdga_load_font("assets/Inter.ttf", 18.0f);
if (!f) return 1;

sdui_set_font(f);   // подключить к UI-рендереру

// ... использовать ...

sdui_set_font(NULL);   // вернуть встроенный
sdga_free_font(f);

Встроенный шрифт

sdga_font *builtin = sdga_load_builtin_font(14.0f);
sdui_set_font(builtin);

Множественные размеры (faces)

Один шрифт может содержать несколько растеризаций под разные размеры. Все глифы живут в одном атласе — UI остаётся в одном draw call.

sdga_font_add_face(f, 14.0f);
sdga_font_add_face(f, 20.0f);
sdga_font_add_face(f, 32.0f);

// UI автоматически выберет ближайший:
sdui_text(x, y, "Title", 28.0f, color);  // возьмёт face 32

Метрики

float asc  = sdga_font_ascent(f);        // над baseline
float desc = sdga_font_descent(f);       // под baseline (положительное)
float lh   = sdga_font_line_height(f);   // полная высота строки

float w = sdui_text_width("Hello", 18.0f);
float h = sdui_text_height(18.0f);

Доступ к глифам

int gid = sdga_font_glyph_index(f, 'A');
const sdga_glyph *g = sdga_font_face_glyph(f, 0, 'A');

g->w, g->h;               // размер bitmap
g->bearing_x, g->bearing_y;
g->advance;
g->u0, g->v0, g->u1, g->v1;   // UV в атласе
Не нужно растеризовать самому Обычно достаточно sdga_load_font + sdui_set_font. UI-рендерер сам подбирает face и рисует глифы.

Аудио-файлы

Декодеры WAV и MP3 → interleaved float32 [-1, 1]. Это файлы — для проигрывания см. Spatial-аудио.

WAV

sdga_wav wav;
if (sdga_load_wav("sound.wav", &wav) == 0) {
    // wav.channels    — 1 или 2
    // wav.sample_rate — Гц
    // wav.frame_count — фреймов (= samples / channels)
    // wav.samples     — interleaved float*
    sdga_free_wav(&wav);
}

Поддержка: PCM u8/s16/s24/s32, IEEE float32, WAVE_FORMAT_EXTENSIBLE, mono/stereo.

MP3

sdga_mp3 mp3;
if (sdga_load_mp3("music.mp3", &mp3) == 0) {
    // mp3.channels, mp3.sample_rate, mp3.frame_count, mp3.samples
    sdga_free_mp3(&mp3);
}

MPEG 1/2/2.5 Layer III. Формат результата — тот же, что у WAV.

Универсальная загрузка

Для звуковой подсистемы движка:

sdge_audio_buffer *buf = sdge_audio_load_file("sound.mp3");
// сам выберет декодер по расширению
// и ресемплирует в 48 kHz при загрузке

Рендерер

Forward PBR-lite, OpenGL 3.3+. Один шейдер для всего 3D: ambient + sun + до 8 point lights, скиннинг, нормал-маппинг.

Инициализация

sdgr_init();                // после make_context_current
sdgr_resize(w, h);          // при изменении размера
// ...
sdgr_shutdown();

Кадр

sdgr_begin_frame((float[4]){0.09f, 0.10f, 0.14f, 1.0f});
// ... draw calls
sdgr_end_frame();

Камера

sdgr_set_camera(cam.view, cam.proj, cam.pos);

// Проверка видимости сферы (frustum culling)
if (sdgr_sphere_visible(center, radius))
    draw_mesh(...);

float cam_pos[3];
sdgr_get_camera_pos(cam_pos);

Освещение

sdgr_set_ambient(0.2f, 0.2f, 0.25f);
sdgr_set_sun((float[3]){-0.5f,-1.0f,-0.3f}, (float[3]){1,1,1}, 2.0f);

sdgr_set_point_light(0,
    (float[3]){2, 3, 0},
    (float[3]){1, 0.8f, 0.6f},
    1.5f,       // intensity
    10.0f);     // radius

До SDGR_MAX_POINT_LIGHTS (8) источников. Если задать индекс больше текущего максимума, счётчик увеличится.

Отрисовка меша

Простая версия — только base color и один источник света:

sdgr_draw_mesh(&mesh, model_matrix,
               NULL, 0,               // joints, joint_count (NULL для статики)
               (float[4]){0.8f, 0.8f, 0.8f, 1.0f},
               0.0f, 0.6f,            // metallic, roughness
               0,                     // double_sided
               0);                    // base_texture (0 = без текстуры)

Полная версия — все 5 слотов, normal map, alpha mode:

sdgr_material mat = {0};
mat.base_color[0..3] = ...;
mat.metallic         = 0.2f;
mat.roughness        = 0.6f;
mat.alpha_mode       = SDGR_ALPHA_OPAQUE;
mat.normal_scale     = 1.0f;
// uv_set / uv_offset / uv_scale / uv_rot для каждой текстуры

sdgr_texture_set tex = {
    .base_color          = base_tex,
    .metallic_roughness  = mr_tex,

Модели

Высокоуровневый субъект: загрузка glTF, анимации, иерархия нод, отрисовка со скиннингом и frustum-culling.

Загрузка и отрисовка

sdge_model *hero = sdge_model_load("assets/hero.glb");
if (!hero) return 1;

// в цикле:
sdge_model_draw(hero, NULL);   // NULL = identity transform

// в конце:
sdge_model_destroy(hero);
Что делает sdge_model_load Загружает glTF, загружает все меши на GPU, декодирует и грузит текстуры (sRGB для base/emissive, linear для остальных), создаёт материалы.

Анимации

int n = sdge_model_anim_count(hero);
for (int i = 0; i < n; ++i)
    printf("anim %d: %s\n", i, sdge_model_anim_name(hero, i));

sdge_model_play(hero, 0, 1);       // anim 0, loop
sdge_model_set_time(hero, 2.5f);   // прыгнуть в середину

// в цикле
sdge_model_advance(hero, dt);

// остановить
sdge_model_stop(hero);

Множественная отрисовка

Модель можно нарисовать несколько раз с разной трансформацией:

sdg_mat4 world;
m4_translation(world, 5, 0, 0);
sdge_model_draw(hero, world);   // копия на (5, 0, 0)

Иерархия (для редактора/вьюера)

int count = sdge_model_node_count(hero);
for (int i = 0; i < count; ++i) {
    const char *name   = sdge_model_node_name(hero, i);
    int parent         = sdge_model_node_parent(hero, i);
    int mesh_idx       = sdge_model_node_mesh(hero, i);

    float t[3], r[4], s[3];
    sdge_model_node_get_transform(hero, i, t, r, s);

    sdg_mat4 world;
    sdge_model_node_world(hero, i, world);
}

Изменение трансформа

sdg_vec3 t = {1, 0, 0};
sdg_quat r; q_identity(r);
sdg_vec3 s = {1, 1, 1};

sdge_model_node_set_transform(hero, node_idx, t, r, s);
sdge_model_refresh(hero);   // после батча изменений

Материалы и текстуры

int mat_idx = sdge_model_node_material(hero, node, prim_idx);
int prim_count = sdge_model_node_prim_count(hero, node);

Override текстур

unsigned custom = sdgr_upload_texture(w, h, pixels);

// ВНИМАНИЕ: модель забирает ownership текстуры.
// Не вызывайте sdgr_free_texture сами.
sdge_model_set_node_texture(hero, node, SDGR_SLOT_BASE, custom);

// Проверить, что стоит
unsigned current = sdge_model_get_node_texture(hero, node, SDGR_SLOT_BASE);

// Убрать override (вернуть материал)
sdge_model_set_node_texture(hero, node, SDGR_SLOT_BASE, 0);
Ownership текстуры После sdge_model_set_node_texture текстура принадлежит модели. Она будет освобождена при sdge_model_destroy. Если передадите ту же текстуру в другой слот — будет double-free.

Камера 3D

Орбитальная или FPS-камера с поддержанием view/proj/frustum в актуальном состоянии.

Орбитальный режим

sdge_camera3d cam;
sdge_camera3d_init(&cam, 60.0f, 16.0f/9.0f, 0.1f, 100.0f);

sdge_camera3d_orbit(&cam,
    (float[3]){0, 1, 0},   // центр
    5.0f,                  // расстояние
    0.3f,                  // yaw (радианы)
    0.4f);                 // pitch (радианы, [-1.5, 1.5])

sdge_camera3d_update(&cam);
sdgr_set_camera(cam.view, cam.proj, cam.pos);

Управление от мыши

static float yaw = 0.3f, pitch = 0.4f, dist = 5.0f;

void on_mouse(sdge_window *w, int btn, int action, int mods) {
    dragging = (action == SDGE_PRESS);
}

void on_cursor(sdge_window *w, double x, double y) {
    if (!dragging) { last_x = x; last_y = y; return; }
    yaw   += (float)(x - last_x) * 0.005f;
    pitch -= (float)(y - last_y) * 0.005f;
    pitch = clampf(pitch, -1.4f, 1.4f);
    last_x = x; last_y = y;
    sdge_camera3d_orbit(&cam, center, dist, yaw, pitch);
    sdge_camera3d_update(&cam);
}

FPS-режим

sdge_camera3d_fps(&cam,
    (float[3]){0, 1.6f, 5},   // позиция
    yaw, pitch);              // forward = -Z при (0,0)
sdge_camera3d_update(&cam);

Look-at

sdge_camera3d_look_at(&cam,
    (float[3]){3, 2, 5},      // eye
    (float[3]){0, 0, 0});     // target
sdge_camera3d_update(&cam);

Дополнительно

sdge_camera3d_set_aspect(&cam, w / (float)h);
sdge_camera3d_update(&cam);

float fwd[3];
sdge_camera3d_forward(&cam, fwd);

if (sdge_camera3d_sphere_visible(&cam, center, radius)) {
    // в кадре
}
Не забудьте sdge_camera3d_update После любого изменения полей камеры (или после orbit/fps/look_at) вызовите update. view/proj/frustum обновятся.

Освещение

Три типа: ambient, sun (directional) и point light. Управляются через рендерер.

Ambient

sdge_ambient_light amb;

Sun

    (float[3]){-0.5f, -1.0f, -0.3f},   // направление, куда свет ЛЕТИТ
    (float[3]){1, 1, 1},
    2.0f);                              // intensity
sdge_sun_apply(&sun);

Point light

sdge_point_light pl;

    (float[3]){2, 3, 0},               // позиция

Плеер анимаций

Низкоуровневый API для скелетной анимации. Обычно хватает sdge_model_play, но если нужен контроль — вот он.

Жизненный цикл

// Один проигрыватель = одна модель + одна анимация
sdge_anim_player *p = sdge_anim_player_create(gltf_model, 0);
// ... 
sdge_anim_player_destroy(p);

Управление временем

sdge_anim_player_advance(p, dt);
sdge_anim_player_set_time(p, 0.5f);
sdge_anim_player_reset(p);            // time → 0

float t = sdge_anim_player_get_time(p);
float dur = sdge_anim_player_get_duration(p);

sdge_anim_player_set_looping(p, 1);
int loop = sdge_anim_player_is_looping(p);

Матрицы и skin-палитра

// Локальная матрица ноды
sdg_mat4 local;
sdge_anim_player_get_local(p, node_idx, local);

// Мировая матрица (вся цепочка родителей)
sdg_mat4 world;
sdge_anim_player_get_world(p, node_idx, world);

// Палитра для скиннинга
const float *palette = sdge_anim_player_skin_palette(p, &gltf->skins[0]);
// palette = joint_count × mat4 column-major
// готово к glUniformMatrix4fv(loc, joint_count, GL_FALSE, palette)

Оптимизации под капотом

  • Каналы разрешаются один раз при создании плеера
  • Топологический порядок нод — обход линейный, без рекурсии
  • Матрицы ленивые: set_time дважды подряд бесплатно
  • Skin-палитра переиспользует закэшированный world-проход
  • Ключи ищутся бинарным поиском
Обычно не нужен напрямую sdge_model_play(hero, 0, 1) + sdge_model_advance(hero, dt) покрывают 90% случаев. Плеер напрямую — если рисуете несколько экземпляров одной модели с разной анимацией.

Spatial-аудио

Свой software-микшер: 48 kHz stereo, 5 мс блоки, до 64 источников. Inverse-distance, equal-power pan, Doppler, LPF.

Инициализация

sdge_audio_init();
// ...
sdge_audio_terminate();

Если устройства нет (WSL, headless) — движок продолжает работать, но без звука. Проверить:

if (!sdge_audio_has_device())
    fprintf(stderr, "audio: silent mode\n");

Буферы

sdge_audio_buffer *b = sdge_audio_load_wav("step.wav");
// или
sdge_audio_buffer *b = sdge_audio_load_mp3("music.mp3");
// или — универсально, по расширению:
sdge_audio_buffer *b = sdge_audio_load_file("sound.ogg");  // выберет декодер

// ...
sdge_audio_free_buffer(b);

Буфер загружается и ре-семплируется в 48 kHz один раз. Один буфер может использоваться несколькими источниками.

Источники

sdge_audio_source *s = sdge_audio_source_create(buffer);

sdge_audio_source_set_volume(s, 0.8f);
sdge_audio_source_set_loop(s, 1);
sdge_audio_source_set_pitch(s, 1.0f);   // 1 = нормальная высота

sdge_audio_source_play(s);

// ...

sdge_audio_source_stop(s);      // сбросить в начало
sdge_audio_source_pause(s);     // запомнить позицию

sdge_audio_source_destroy(s);

Пространственное позиционирование

sdge_audio_source_set_spatial(s, 1);
sdge_audio_source_set_position(s, 3.0f, 0.0f, -2.0f);
sdge_audio_source_set_velocity(s, 0.0f, 0.0f, 0.0f);

// Модель затухания:
sdge_audio_source_set_distance(s,
    1.0f,    // ref — громкость 100%
    50.0f);  // max — полностью затихнет

Слушатель

sdge_audio_listener_position(cam.pos[0], cam.pos[1], cam.pos[2]);

// forward + up
sdge_audio_listener_orientation(
    0, 0, -1,   // forward
    0, 1, 0);   // up

// Для Doppler
sdge_audio_listener_velocity(vx, vy, vz);

Автоматическая синхронизация с камерой

// Каждый кадр

Физика 2D

Rigid body: круги и OBB, sequential impulses, friction, sleep. Детерминированная — можно тестировать headless.

Мир

sdph2_world *w = sdph2_world_create();
sdph2_world_set_gravity(w, 0.0f, -9.81f);
sdph2_world_set_damping(w, 0.02f, 0.05f);
sdph2_world_set_iterations(w, 8);

// Границы мира (стены)
sdph2_world_set_bounds(w, 1, -20, -20, 20, 20);

Тела

sdph2_body *ball = sdph2_add_circle(w,
    0.0f, 5.0f,    // x, y
    0.5f,          // radius
    1.0f);         // mass (>0 = dynamic, <=0 = static)

sdph2_body *floor = sdph2_add_box(w,
    0.0f, -0.5f,   // center
    10.0f, 0.5f,   // half extents
    0.0f,          // mass (0 = static)
    0.0f);         // angle (rad)
Стабильные указатели sdph2_body* не меняется при добавлении других тел. Можно смело хранить в своих структурах.

Шаг симуляции

sdph2_step(w, 1.0f / 60.0f);

// Прочитать состояние
for (int i = 0; i < w->count; ++i) {
    sdph2_body *b = w->bodies[i];
    draw_at(b->pos[0], b->pos[1], b->angle);
}

Управление телом

sdph2_set_velocity(ball, 5.0f, 0.0f);
sdph2_set_angular_velocity(ball, 2.0f);
sdph2_set_position(ball, 0.0f, 10.0f);
sdph2_set_angle(ball, 0.5f);

sdph2_apply_impulse(ball,
    10.0f, 0.0f,    // impulse (ix, iy)
    0.0f, 0.5f);    // offset от центра

sdph2_set_restitution(ball, 0.7f);   // упругость
sdph2_set_friction(ball, 0.3f);

Сон

Тела автоматически засыпают, если долго не двигаются. Разбудить:

sdph2_wake(body);
sdph2_sleep(body);

Запросы

// Какие тела перекрывают круг?

// Точка внутри?
if (sdph2_point_in_body(body, mouse_x, mouse_y))
    select(body);

// Суммарная кинетическая энергия (проверка стабилизации)
float e = sdph2_energy(w);

Физика 3D

Полный аналог 2D: сферы и OBB, 15-осевой SAT, диагональный тензор инерции, quaternion-ориентация.

Мир

sdph3_world *w = sdph3_world_create();
sdph3_world_set_gravity(w, 0.0f, -9.81f, 0.0f);
sdph3_world_set_iterations(w, 12);

sdph3_world_set_bounds(w, 1, -20, -20, -20, 20, 20, 20);

Тела

sdph3_body *ball = sdph3_add_sphere(w,
    0, 5, 0,     // x, y, z
    0.5f,        // radius
    1.0f);       // mass

sdph3_body *ground = sdph3_add_box(w,
    0, -0.5f, 0,
    10.0f, 0.5f, 10.0f,
    0.0f);       // mass = 0 → static

sdph3_body *cube = sdph3_add_box(w, 0, 3, 0, 0.5f, 0.5f, 0.5f, 1.0f);

Шаг

sdph3_step(w, 1.0f / 60.0f);

// Отрисовать все тела
for (int i = 0; i < w->count; ++i) {
    sdph3_body *b = w->bodies[i];

    sdg_mat4 model;
    m4_from_trs(model, b->pos, b->quat, (sdg_vec3){1,1,1});

    // ...
}

Ориентация

Источник истины — кватернион. Euler — только для чтения/записи удобным способом.

// Задать через Euler (Rz·Ry·Rx)
sdph3_set_angle(body, 0.0f, 0.5f, 0.0f);

// Или напрямую quaternion
float q[4] = {0, 0.7071f, 0, 0.7071f};
sdph3_set_quat(body, q);

// Прочитать
float out[4];
sdph3_get_quat(body, out);

Импульс

// Толчок в точку (rx, ry, rz) относительно центра масс
sdph3_apply_impulse(body,
    10.0f, 0.0f, 5.0f,    // impulse
    0.5f, 0.5f, 0.0f);    // offset

Запросы

sdph3_body *hits[16];
int n = sdph3_query_sphere(w, x, y, z, radius, hits, 16);

if (sdph3_point_in_body(body, x, y, z))
    select(body);

float e = sdph3_energy(w);   // стабильность

Типы

Аналогично 2D: SDPH3_STATIC, SDPH3_DYNAMIC, SDPH3_KINEMATIC.

Что под капотом

  • SAT по 15 осям: 6 граней + 9 рёбер
  • Face manifold: incident face клиппится reference face → до 4 контактных точек (стабильные стопки)
  • Sequential impulses с restitution и friction
  • Split impulse: velocity bias отдельно от position correction (нет «выстреливания» из стены)
  • Диагональный тензор инерции: длинный бокс вращается вокруг X и Z по-разному
  • Safety caps: max линейная/угловая скорость на выходе шага

Менеджер сцен

Переключение между сценами с fade-переходом. Сцена — просто struct с колбэками.

Структура сцены

typedef struct sdge_scene {
    const char *name;

    void (*on_enter )(sdge_scene *s);
    void (*on_exit  )(sdge_scene *s);
    void (*on_update)(sdge_scene *s, float dt);
    void (*on_render)(sdge_scene *s);              // 3D
    void (*on_gui   )(sdge_scene *s, sdge_gui *g); // UI
    void (*on_event )(sdge_scene *s, int type, int param);

    void *user;
} sdge_scene;

Определение сцены

static void menu_enter(sdge_scene *s) { /* загрузка ресурсов */ }
static void menu_exit (sdge_scene *s) { /* освобождение */ }
static void menu_update(sdge_scene *s, float dt) { /* логика */ }
static void menu_render(sdge_scene *s) { /* 3D */ }
static void menu_gui(sdge_scene *s, sdge_gui *g) { /* UI */ }

sdge_scene menu = {
    .name      = "menu",
    .on_enter  = menu_enter,
    .on_exit   = menu_exit,
    .on_update = menu_update,
    .on_render = menu_render,
    .on_gui    = menu_gui,
};

Менеджер

sdge_scene_manager mgr;
sdge_scene_mgr_init(&mgr);

sdge_scene_mgr_add(&mgr, &menu);
sdge_scene_mgr_add(&mgr, &world);
sdge_scene_mgr_switch_now(&mgr, 0);   // мгновенно

Кадр

sdge_scene_mgr_update(&mgr, dt);
sdge_scene_mgr_render(&mgr);

sdge_gui_begin(gui, w, h);
sdge_scene_mgr_render_gui(&mgr, gui);   // рисует и fade
sdge_gui_end(gui);

Переход с fade

sdge_scene_mgr_switch(&mgr, 1);   // 0.45 сек fade out → fade in

// Настройка:
mgr.fade_speed = 0.3f;   // секунд

sdge_scene_mgr_set_fade_color(&mgr, (float[4]){0, 0, 0, 1});

// Проверить статус:
if (sdge_scene_mgr_transitioning(&mgr))
    disable_input();

float t = sdge_scene_mgr_fade(&mgr);   // 0..1
